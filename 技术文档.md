## 1. 技术选型

### 1.1 核心技术栈

| 类别 | 技术 | 版本 | 用途 |
|------|------|------|------|
| 结构 | HTML5 | - | 页面骨架 |
| 样式 | CSS3 + CSS 变量 | - | 主题切换、响应式 |
| 逻辑 | 原生 JavaScript (ES6+) | - | 交互逻辑（无框架依赖） |
| 3D 渲染 | Three.js | r160+ | 3D 立方体 Hero |
| 粒子系统 | 原生 Canvas 2D | - | 像素粒子雨 + 星空 |
| 平滑滚动 | Lenis (CDN) | latest | 平滑滚动体验 |
| 图标 | Lucide Icons (SVG) | - | UI 图标 |
| 打包 | 无（单文件） | - | 导出时内联所有资源 |

### 1.2 选型理由
- **纯前端无框架**: 满足"纯前端实现"要求，避免构建工具链，降低复杂度
- **Three.js**: 业界标准 WebGL 库，实现 Minecraft 风格 3D 立方体
- **原生 Canvas 粒子**: 比 Three.js 粒子更轻量，像素粒子雨用 2D Canvas 即可，性能更好
- **Lenis**: 轻量级平滑滚动库，CDN 引入，提供丝滑滚动
- **CSS 变量**: 实现 5 套主题一键切换，无需切换 class 链路
- **localStorage**: 零后端依赖的持久化方案

### 1.3 CDN 依赖（导出时内联）
```html
<script src="https://cdn.jsdelivr.net/npm/three@0.160.0/build/three.min.js"></script>
<script src="https://cdn.jsdelivr.net/npm/@studio-freight/lenis@1.0.42/dist/lenis.min.js"></script>
<link href="https://cdn.jsdelivr.net/npm/lucide-static@0.400.0/font/lucide.css" rel="stylesheet">
```
> 导出单文件时，Three.js / Lenis 的源码将内联进 HTML，离线可用。

## 4. 核心模块设计

### 4.1 路由系统 (router.js)

采用 Hash 路由，无需服务器配置：

```javascript
class Router {
  constructor(routes) {
    this.routes = routes;
    this.current = "home";
  }
  
  init() {
    window.addEventListener("hashchange", () => this.handleRoute());
    this.handleRoute();
  }
  
  handleRoute() {
    const hash = window.location.hash.slice(1) || "home";
    this.navigate(hash);
  }
  
  navigate(pageId) {
    this.current = pageId;
    // 隐藏所有页面，显示目标页面
    document.querySelectorAll("[data-page]").forEach(p => {
      p.classList.toggle("active", p.dataset.page === pageId);
    });
    // 更新导航高亮
    document.querySelectorAll("[data-nav]").forEach(n => {
      n.classList.toggle("active", n.dataset.nav === pageId);
    });
    window.scrollTo({ top: 0, behavior: "smooth" });
  }
}
```

### 4.2 数据存储 (storage.js)

```javascript
const STORAGE_KEY = "pixelverse_portfolio_data";

const Storage = {
  // 加载配置（localStorage → 合并默认值）
  load() {
    try {
      const saved = localStorage.getItem(STORAGE_KEY);
      if (saved) {
        return deepMerge(defaultConfig, JSON.parse(saved));
      }
    } catch (e) { console.warn("加载配置失败", e); }
    return JSON.parse(JSON.stringify(defaultConfig));
  },
  
  // 保存配置（防抖）
  save: debounce((config) => {
    try {
      localStorage.setItem(STORAGE_KEY, JSON.stringify(config));
    } catch (e) { console.warn("保存失败", e); }
  }, 500),
  
  // 恢复默认
  reset() {
    localStorage.removeItem(STORAGE_KEY);
    return JSON.parse(JSON.stringify(defaultConfig));
  },
  
  // 导出 JSON
  exportJSON(config) {
    const blob = new Blob([JSON.stringify(config, null, 2)], { type: "application/json" });
    downloadBlob(blob, "portfolio-config.json");
  },
  
  // 导入 JSON
  importJSON(file) {
    return new Promise((resolve, reject) => {
      const reader = new FileReader();
      reader.onload = e => resolve(JSON.parse(e.target.result));
      reader.onerror = reject;
      reader.readAsText(file);
    });
  }
};
```

### 4.4 粒子系统 (particles.js)

基于 Canvas 2D 实现像素粒子雨 + 星空：

```javascript
class ParticleSystem {
  constructor(canvas) {
    this.canvas = canvas;
    this.ctx = canvas.getContext("2d");
    this.raindrops = [];
    this.stars = [];
    this.config = { rainCount: 80, starCount: 120, speed: 1, color: "#D8C2D6" };
  }
  
  init() {
    this.resize();
    window.addEventListener("resize", () => this.resize());
    this.createRain();
    this.createStars();
    this.animate();
  }
  
  createRain() {
    this.raindrops = [];
    for (let i = 0; i < this.config.rainCount; i++) {
      this.raindrops.push({
        x: Math.random() * this.canvas.width,
        y: Math.random() * this.canvas.height,
        size: Math.random() * 3 + 2,  // 像素方块大小
        speed: Math.random() * 1.5 + this.config.speed,
        opacity: Math.random() * 0.5 + 0.2
      });
    }
  }
  
  createStars() {
    this.stars = [];
    for (let i = 0; i < this.config.starCount; i++) {
      this.stars.push({
        x: Math.random() * this.canvas.width,
        y: Math.random() * this.canvas.height,
        size: Math.random() * 2 + 1,
        twinkleSpeed: Math.random() * 0.02 + 0.005,
        phase: Math.random() * Math.PI * 2
      });
    }
  }
  
  animate() {
    const ctx = this.ctx;
    ctx.clearRect(0, 0, this.canvas.width, this.canvas.height);
    
    // 绘制星星
    this.stars.forEach(star => {
      star.phase += star.twinkleSpeed;
      const opacity = 0.3 + Math.sin(star.phase) * 0.3;
      ctx.fillStyle = `rgba(255,255,255,${opacity})`;
      ctx.fillRect(star.x, star.y, star.size, star.size);
    });
    
    // 绘制像素粒子雨（硬边方块，无抗锯齿）
    ctx.imageSmoothingEnabled = false;
    this.raindrops.forEach(drop => {
      drop.y += drop.speed;
      if (drop.y > this.canvas.height) {
        drop.y = -drop.size;
        drop.x = Math.random() * this.canvas.width;
      }
      ctx.fillStyle = hexToRgba(this.config.color, drop.opacity);
      ctx.fillRect(Math.floor(drop.x), Math.floor(drop.y), drop.size, drop.size);
    });
    
    requestAnimationFrame(() => this.animate());
  }
  
  setColor(color) { this.config.color = color; }
  setConfig(config) { Object.assign(this.config, config); this.createRain(); this.createStars(); }
}
```

### 4.5 3D 立方体 Hero (hero3d.js)

使用 Three.js 实现 Minecraft 风格立方体：

```javascript
class Hero3D {
  constructor(container) {
    this.container = container;
    this.scene = new THREE.Scene();
    this.camera = new THREE.PerspectiveCamera(50, container.clientWidth / container.clientHeight, 0.1, 1000);
    this.renderer = new THREE.WebGLRenderer({ alpha: true, antialias: false });
    this.group = new THREE.Group();
  }
  
  init() {
    this.renderer.setSize(this.container.clientWidth, this.container.clientHeight);
    this.renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
    this.container.appendChild(this.renderer.domElement);
    
    this.camera.position.set(0, 0, 8);
    
    // 创建立方体组合（类似 Minecraft 方块堆叠）
    const cubeGeo = new THREE.BoxGeometry(1, 1, 1);
    const colors = [0xD8C2D6, 0xB8CDE0, 0xFFFFFF, 0xC8C4D0];
    
    // 3x3x3 立方体阵列
    for (let x = -1; x <= 1; x++) {
      for (let y = -1; y <= 1; y++) {
        for (let z = -1; z <= 1; z++) {
          if (Math.random() > 0.3) {  // 随机空缺，形成镂空
            const mat = new THREE.MeshLambertMaterial({ 
              color: colors[Math.floor(Math.random() * colors.length)]
            });
            const cube = new THREE.Mesh(cubeGeo, mat);
            cube.position.set(x * 1.05, y * 1.05, z * 1.05);
            cube.userData.originalY = cube.position.y;
            this.group.add(cube);
          }
        }
      }
    }
    this.scene.add(this.group);
    
    // 光照
    const ambient = new THREE.AmbientLight(0xffffff, 0.6);
    this.scene.add(ambient);
    const dir = new THREE.DirectionalLight(0xffffff, 0.8);
    dir.position.set(5, 5, 5);
    this.scene.add(dir);
    
    this.addInteraction();
    this.animate();
  }
  
  addInteraction() {
    let isDragging = false, prevX = 0, prevY = 0;
    this.container.addEventListener("mousedown", e => { isDragging = true; prevX = e.clientX; prevY = e.clientY; });
    window.addEventListener("mouseup", () => isDragging = false);
    window.addEventListener("mousemove", e => {
      if (!isDragging) return;
      const dx = e.clientX - prevX, dy = e.clientY - prevY;
      this.group.rotation.y += dx * 0.01;
      this.group.rotation.x += dy * 0.01;
      prevX = e.clientX; prevY = e.clientY;
    });
    // 触摸支持
    this.container.addEventListener("touchstart", e => { isDragging = true; prevX = e.touches[0].clientX; prevY = e.touches[0].clientY; });
    window.addEventListener("touchend", () => isDragging = false);
    window.addEventListener("touchmove", e => {
      if (!isDragging) return;
      const dx = e.touches[0].clientX - prevX, dy = e.touches[0].clientY - prevY;
      this.group.rotation.y += dx * 0.01;
      this.group.rotation.x += dy * 0.01;
      prevX = e.touches[0].clientX; prevY = e.touches[0].clientY;
    });
  }
  
  animate() {
    requestAnimationFrame(() => this.animate());
    // 自动缓慢旋转
    if (!this.isDragging) {
      this.group.rotation.y += 0.003;
    }
    // 方块浮动动画
    this.group.children.forEach((cube, i) => {
      cube.position.y = cube.userData.originalY + Math.sin(Date.now() * 0.001 + i) * 0.1;
    });
    this.renderer.render(this.scene, this.camera);
  }
}
```

```javascript
    // 移除之前的高亮

    this.bindFormEvents(path);

    this.isOpen = !this.isOpen;
    this.panel.classList.toggle("open", this.isOpen);
    document.body.classList.toggle("workshop-open", this.isOpen);
  }
  
  reset() {
    if (confirm("确定恢复所有内容为默认值？")) {
      siteConfig = Storage.reset();
      location.reload();
    }
  }
}
```

```javascript
const FormBuilder = {
```

### 4.8 单文件 HTML 导出器 (exporter.js)

```javascript
const Exporter = {
  async export() {
    // 1. 获取当前完整 HTML
    let html = document.documentElement.outerHTML;
    
    // 2. 内联所有 CSS
    const styles = Array.from(document.querySelectorAll("link[rel='stylesheet']"));
    for (const link of styles) {
      const css = await fetch(link.href).then(r => r.text());
      html = html.replace(link.outerHTML, `<style>${css}</style>`);
    }
    
    // 3. 内联所有 JS (Three.js, Lenis 等)
    const scripts = Array.from(document.querySelectorAll("script[src]"));
    for (const script of scripts) {
      const js = await fetch(script.src).then(r => r.text());
      html = html.replace(script.outerHTML, `<script>${js}<\/script>`);
    }
    
    // 4. 将所有图片转为 base64 内联
    const images = document.querySelectorAll("img");
    for (const img of images) {
      if (img.src.startsWith("http") || img.src.startsWith("data:")) continue;
      const base64 = await this.imageToBase64(img.src);
      img.src = base64;
    }
    
    // 5. 注入当前配置（覆盖默认值）
    const configScript = `<script>window.__PORTFOLIO_CONFIG__ = ${JSON.stringify(siteConfig)};<\/script>`;
    html = html.replace("</head>", configScript + "</head>");
    
    // 6. 移除工坊相关元素（导出后不需要编辑功能）
    html = html.replace(/<aside id="workshop-panel"[\s\S]*?<\/aside>/, "");
    html = html.replace(/<button id="workshop-toggle"[\s\S]*?<\/button>/, "");
    
    // 7. 下载
    const blob = new Blob([html], { type: "text/html;charset=utf-8" });
    downloadBlob(blob, "portfolio.html");
  },
  
  async imageToBase64(src) {
    return new Promise((resolve) => {
      const img = new Image();
      img.crossOrigin = "anonymous";
      img.onload = () => {
        const canvas = document.createElement("canvas");
        canvas.width = img.width;
        canvas.height = img.height;
        canvas.getContext("2d").drawImage(img, 0, 0);
        resolve(canvas.toDataURL("image/png"));
      };
      img.src = src;
    });
  }
};
```

## 6. 性能优化策略

| 优化点 | 方案 |
|--------|------|
| 3D 渲染 | `antialias: false`（像素风不需要抗锯齿），`pixelRatio` 限制为 2 |
| 粒子系统 | 使用 `imageSmoothingEnabled = false`，粒子数量根据设备性能自适应 |
| 图片 | 导出时压缩，限制最大尺寸 1920px，WebP 优先 |
| 滚动动画 | 使用 `IntersectionObserver` 触发入场动画，非 scroll 事件监听 |
| 重绘 | 粒子 Canvas 使用 `requestAnimationFrame`，页面不可见时暂停 |
| 存储 | localStorage 配置对象精简，图片用 base64 时限制大小 |

---

## 7. 安全与边界

- localStorage 容量约 5MB，超大图片需提示用户压缩
- 导出单文件时，Three.js 源码约 600KB，内联后文件较大但可接受
- 工坊编辑仅在本地操作，不涉及网络请求，无 XSS 风险（用户编辑自己的内容）
- 视频文件不内联到单文件（过大），导出时提示保留视频为外部文件或链接
